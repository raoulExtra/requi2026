0027 def sparql(query: str) -> list[dict[str, str]]:
0028     request = Request(
0029         SPARQL_ENDPOINT,
0030         data=urlencode({"query": query, "format": "json"}).encode(),
0031         headers={
0032             "Accept": "application/sparql-results+json",
0033             "Content-Type": "application/x-www-form-urlencoded",
0034             "User-Agent": USER_AGENT,
0035         },
0036         method="POST",
0037     )
0038     for attempt in range(4):
0039         try:
0040             with urlopen(request, timeout=60) as response:
0041                 bindings = json.load(response)["results"]["bindings"]
0042             return [{key: value["value"] for key, value in row.items()} for row in bindings]
0043         except HTTPError as error:
0044             if error.code not in RETRYABLE or attempt == 3:
0045                 raise
0046             retry_after = error.headers.get("Retry-After")
0047             time.sleep(float(retry_after) if retry_after else 2**attempt)
0048     return []

0055 def find_lexemes(lemmas: list[str], language: str, category: str) -> dict[str, str]:
0056     values = " ".join(literal(lemma, language) for lemma in sorted(set(lemmas)))
0057     query = f"""SELECT ?lemma ?lexeme WHERE {{
0058       VALUES ?lemma {{ {values} }}
0059       ?lexeme a ontolex:LexicalEntry ; wikibase:lemma ?lemma ;
0060                dct:language wd:{LANGUAGES[language]} ;
0061                wikibase:lexicalCategory wd:{LEXICAL_CATEGORIES[category]} .
0062     }}"""
0063     return {row["lemma"]: row["lexeme"].rsplit("/", 1)[-1] for row in sparql(query)}

0066 def find_items(labels: list[str]) -> dict[str, tuple[str, str]]:
0067     values = " ".join(literal(label) for label in sorted(set(labels)))
0068     query = f"""SELECT ?label ?item ?description WHERE {{
0069       VALUES ?label {{ {values} }}
0070       ?item rdfs:label ?label .
0071       OPTIONAL {{ ?item schema:description ?description . FILTER(LANG(?description) = "en") }}
0072     }}"""
0073     matches: dict[str, tuple[str, str]] = {}
0074     for row in sparql(query):
0075         matches.setdefault(row["label"], (row["item"].rsplit("/", 1)[-1], row.get("description", "")))
0076     return matches
