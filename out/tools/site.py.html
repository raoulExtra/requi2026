0018 def href(target: str, source: Path) -> str:
0019     if target.startswith(("http://", "https://", "#", "mailto:")):
0020         return target
0021     anchor = ""
0022     if "#" in target:
0023         target, anchor = target.split("#", 1)
0024     resolved = (source.parent / target).resolve()
0025     target_rel = resolved.relative_to(ROOT)
0026     target_rel = (target_rel / "index.html") if target.endswith("/") else target_rel.with_suffix(".html") if target.endswith(".md") else target_rel
0027     source_rel = source.relative_to(ROOT)
0028     return Path(__import__("os").path.relpath(target_rel, source_rel.parent)).as_posix() + (f"#{anchor}" if anchor else "")
0029 def inline(text: str, source: Path) -> str:
0030     escaped = html.escape(text, quote=False)
0031     placeholders: list[str] = []
0032 
0033     def replace(match: re.Match[str]) -> str:
0034         label, target = match.groups()
0035         token = f"\x00{len(placeholders)}\x00"
0036         placeholders.append(f'<a href="{html.escape(href(target, source), quote=True)}">{html.escape(label)}</a>')
0037         return token
0038 
0039     escaped = LINK.sub(replace, escaped)
0040     escaped = re.sub(r"`([^`]+)`", r"<code>\1</code>", escaped)
0041     escaped = re.sub(r"\*\*([^*]+)\*\*", r"<strong>\1</strong>", escaped)
0042     for index, value in enumerate(placeholders):
0043         escaped = escaped.replace(f"\x00{index}\x00", value)
0044     return escaped

0047 def render(markdown: str, source: Path) -> str:
0048     lines = markdown.splitlines()
0049     output: list[str] = []
0050     in_code = False
0051     paragraph: list[str] = []
0052     list_open = False
0053     table_rows: list[list[str]] = []
0054 
0055     def flush_paragraph() -> None:
0056         nonlocal paragraph
0057         if paragraph:
0058             output.append(f"<p>{inline(' '.join(paragraph), source)}</p>")
0059             paragraph = []
0060 
0061     def close_list() -> None:
0062         nonlocal list_open
0063         if list_open:
0064             output.append("</ul>")
0065             list_open = False
0066 
0067     def close_table() -> None:
0068         nonlocal table_rows
0069         if not table_rows:
0070             return
0071         output.append("<table><thead><tr>" + "".join(f"<th>{inline(cell, source)}</th>" for cell in table_rows[0]) + "</tr></thead><tbody>")
0072         for row in table_rows[1:]:
0073             output.append("<tr>" + "".join(f"<td>{inline(cell, source)}</td>" for cell in row) + "</tr>")
0074         output.append("</tbody></table>")
0075         table_rows = []
0076 
0077     for line in lines:
0078         if line.startswith("```"):
0079             flush_paragraph()
0080             close_list()
0081             close_table()
0082             output.append("</code></pre>" if in_code else "<pre><code>")
0083             in_code = not in_code
0084             continue
0085         if in_code:
0086             output.append(html.escape(line) + "\n")
0087             continue
0088         if line.strip().startswith("|") and line.strip().endswith("|"):
0089             flush_paragraph()
0090             close_list()
0091             cells = [cell.strip() for cell in line.strip().strip("|").split("|")]
0092             if not all(re.fullmatch(r":?-+:?", cell) for cell in cells):
0093                 table_rows.append(cells)
0094             continue
0095         close_table()
0096         heading = re.match(r"^(#{1,6})\s+(.+)$", line)
0097         item = re.match(r"^\s*[-*]\s+(.+)$", line)
0098         quote = re.match(r"^\s*>\s*(.*)$", line)
0099         if heading:
0100             flush_paragraph()
0101             close_list()
0102             level = len(heading.group(1))
0103             heading_text = heading.group(2)
0104             heading_id = heading_text.split(" — ", 1)[0].lower() if heading_text.startswith("DB-") else re.sub(r"[^a-z0-9]+", "-", heading_text.lower()).strip("-")
0105             output.append(f'<h{level} id="{heading_id}">{inline(heading_text, source)}</h{level}>')
0106         elif item:
0107             flush_paragraph()
0108             if not list_open:
0109                 output.append("<ul>")
0110                 list_open = True
0111             output.append(f"<li>{inline(item.group(1), source)}</li>")
0112         elif quote:
0113             flush_paragraph()
0114             close_list()
0115             output.append(f"<blockquote>{inline(quote.group(1), source)}</blockquote>")
0116         elif line.strip():
0117             close_list()
0118             paragraph.append(line.strip())
0119         else:
0120             flush_paragraph()
0121             close_list()
0122     flush_paragraph()
0123     close_list()
0124     close_table()
0125     return "\n".join(output)
