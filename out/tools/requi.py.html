0146             PRIMARY KEY(command_name, name)
0147         );
0148         CREATE TABLE IF NOT EXISTS command_aliases(command_name TEXT, alias TEXT PRIMARY KEY);
0149         DROP VIEW IF EXISTS v_name_collisions;
0150         DROP VIEW IF EXISTS v_entities;
0151         DROP VIEW IF EXISTS v_commands;

0160         "CREATE VIEW v_commands AS "
0161         "SELECT c.name, c.purpose, c.usage, a.name AS argument, a.flag, a.required, "
0162         "a.value_type, a.description FROM commands c "
0163         "LEFT JOIN command_arguments a ON a.command_name = c.name"
0164     )
0165     db.executescript(
0166         """
0167         CREATE VIEW v_entities AS
0168         SELECT 'term' AS kind, name, path AS source FROM terms
0169         UNION ALL SELECT 'scope', name, 'scopes/' || id || '.md' FROM scopes
0170         UNION ALL SELECT 'layer', id, 'layers/' || id || '.md' FROM layers
0171         UNION ALL SELECT 'db-table', name, 'sqlite table' FROM sqlite_master
0172             WHERE type = 'table' AND name NOT LIKE 'sqlite_%'
0173         UNION ALL SELECT 'command', name, 'commands' FROM commands
0174         UNION ALL SELECT 'view', name, 'sqlite view' FROM sqlite_master WHERE type = 'view';
0175         CREATE VIEW v_name_collisions AS
0176         SELECT lower(name) AS normalized_name, COUNT(*) AS match_count,
0177             group_concat(kind || ':' || source, ' | ') AS matches
0178         FROM v_entities GROUP BY lower(name) HAVING COUNT(*) > 1;

0474     term_paths = {path for (path,) in db.execute("SELECT path FROM terms")}
0475     layer_ids = {layer_id for (layer_id,) in db.execute("SELECT id FROM layers")}
0476     scope_ids_set = {scope_id for (scope_id,) in db.execute("SELECT id FROM scopes")}
0477     requirement_ids = {requirement_id for (requirement_id,) in db.execute("SELECT id FROM requirements")}

0479         for (layer_id,) in db.execute("SELECT layer_id FROM term_layers WHERE term_path = ?", (path,)):
