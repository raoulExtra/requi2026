0031 def _usage_arguments(command: str, usage: str) -> list[dict[str, object]]:
0032     """Read the compact usage grammar used by COMMANDS."""
0033     tail = usage.split(command, 1)[1].strip() if command in usage else ""
0034     pieces = re.findall(r"\[[^\]]+\]|--[A-Za-z0-9_-]+(?:\s+[A-Z][A-Za-z0-9_-]*)?|[A-Z][A-Za-z0-9_-]*", tail)
0035     result: list[dict[str, object]] = []
0036     for piece in pieces:
0037         optional = piece.startswith("[") and piece.endswith("]")
0038         token = piece[1:-1].strip() if optional else piece
0039         if token.startswith("--"):
0040             parts = token.split()
0041             flag = parts[0]
0042             metavar = parts[1] if len(parts) > 1 else None
0043             name = flag.removeprefix("--").replace("-", "_")
0044             result.append({
0045                 "name": name,
0046                 "kind": "option",
0047                 "flag": flag,
0048                 "required": not optional,
0049                 "metavar": metavar,
0050             })
0051         else:
0052             result.append({
0053                 "name": token.lower().replace("-", "_"),
0054                 "kind": "positional",
0055                 "flag": None,
0056                 "required": not optional,
0057                 "metavar": token,
0058             })
0059     return result

0123 def render_tree(commands: tuple[CommandNode, ...]) -> str:
0124     lines: list[str] = []
0125     for index, command in enumerate(commands):
0126         if index:
0127             lines.append("")
0128         lines.append(f"command {command.name}")
0129         lines.append(f"├─ purpose: {command.purpose}")
0130         lines.append(f"├─ usage: {command.usage}")
0131         if command.aliases:
0132             lines.append(f"├─ aliases: {', '.join(command.aliases)}")
0133         lines.append("└─ arguments:")
0134         if not command.arguments:
0135             lines.append("   └─ (none)")
0136             continue
0137         for position, argument in enumerate(command.arguments):
0138             branch = "└─" if position == len(command.arguments) - 1 else "├─"
0139             details = f" [{argument.value_type}]"
0140             if argument.description:
0141                 details += f" — {argument.description}"
0142             lines.append(f"   {branch} {_argument_label(argument)}{details}")
0143     return "\n".join(lines)

0154 def render_argparse(commands: tuple[CommandNode, ...]) -> str:
0155     """Emit parser-construction code; dispatch remains project-specific."""
0156     lines = [
0157         "# Generated by: ./requi.sh ast --format python",
0158         "sub = parser.add_subparsers(dest='command', required=True)",
0159     ]
0160     for command in commands:
0161         aliases = f", aliases={command.aliases!r}" if command.aliases else ""
0162         variable = re.sub(r"\W+", "_", command.name).strip("_") or "command"
0163         lines.append(f"{variable} = sub.add_parser({_python_string(command.name)}, help={_python_string(command.purpose)}{aliases})")
0164         for argument in command.arguments:
0165             if argument.kind == "positional":
0166                 options = [f"metavar={argument.metavar or argument.name.upper()!r}"]
0167                 if not argument.required:
0168                     options.append("nargs='?'")
0169                 lines.append(f"{variable}.add_argument({_python_string(argument.name)}, {', '.join(options)})")
0170                 continue
0171             flag = argument.flag or f"--{argument.name.replace('_', '-')}"
0172             options = [f"dest={argument.name!r}"]
0173             if argument.required:
0174                 options.append("required=True")
0175             if argument.metavar:
0176                 options.append(f"metavar={argument.metavar!r}")
0177             lines.append(f"{variable}.add_argument({_python_string(flag)}, {', '.join(options)})")
0178     return "\n".join(lines)
