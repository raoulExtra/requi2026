0050 def _sections(text: str) -> dict[str, str]:
0051     matches = list(re.finditer(r"^## (.+)$", text, re.M))
0052     return {
0053         match.group(1).strip().lower(): text[match.end(): matches[index + 1].start() if index + 1 < len(matches) else len(text)]
0054         for index, match in enumerate(matches)

0678 def render_database(db: sqlite3.Connection | None = None) -> None:
0679     owns_db = db is None
0680     db = db or sqlite3.connect(ROOT / "requi.db")
0681     sources = database_markdown(db)
0682     for path in (ROOT / "terms").glob("*.md"):
0683         path.unlink()
0684     for path in (ROOT / "scopes").glob("*.md"):
0685         path.unlink()
0686     expected_layers = {ROOT / path for path in sources if path.startswith("layers/") and path != "layers/_layers.md"}
0687     expected_requirements = {ROOT / path for path in sources if path.startswith("requirements/")}
0688     expected_lexemes = {ROOT / path for path in sources if path.startswith("lexemes/L")}
0689     for path in (ROOT / "layers").glob("*.md"):
0690         if path.name != "_layers.md" and path not in expected_layers:
0691             path.unlink()
0692     for path in (ROOT / "requirements").rglob("*.md"):
0693         if path not in expected_requirements:
0694             path.unlink()
0695     for path in (ROOT / "lexemes").glob("L*.md"):
0696         if path not in expected_lexemes:
0697             path.unlink()
0698     for relative, content in sources.items():
0699         if relative.startswith(("terms/", "scopes/")):
0700             continue
0701         target = ROOT / relative
0702         target.parent.mkdir(parents=True, exist_ok=True)
0703         target.write_text(content, encoding="utf-8")
0704     if owns_db:
0705         db.commit()
0706         db.close()
