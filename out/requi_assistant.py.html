0013 ROOT = Path(__file__).resolve().parent
0014 DEFAULT_URL = os.environ.get("LM_STUDIO_URL", "http://barbara:1234/v1")
0015 DEFAULT_MODEL = os.environ.get("LM_MODEL", "nvidia/nemotron-3-nano-4b")
0016 CALL_LOG = ROOT / "last_model_call.md"
0017 # REQUI: DB-001 DB-002 DB-004 ← DB-001 DB-002 DB-004
0018 SYSTEM_PROMPT = """You are a requirements-engineering assistant working on a local Markdown project.
0019 
0020 Project rules:
0021 - Requirements use IDs REQ-### and are never reused or renumbered.
0022 - Each requirement has Status (PROPOSED, ACTIVE, or RETIRED) and Source.
0023 - Terms and scopes are stored in `requi.db`; terms render to `out/terms/<kebab-case-name>.html` and scopes render to `out/scopes/<scope-id>.html`.
0024 - Terms may belong to multiple scopes; scope pages group terms and requirement categories.
0025 - Requirement links to terms and terms link back to requirements.
0026 - Do not invent facts or silently edit files.
0027 
0028 Review the supplied project context. Answer the user's request with concrete file paths,
0029 requirement IDs, and Markdown where useful. If a change is needed, propose it explicitly;
0030 do not claim that files were changed. The assistant itself is read-only.
0031 """

0034 def project_context(max_chars: int) -> str:
0035     files = sorted(
0036         path for path in ROOT.rglob("*")
0037         if path.is_file() and path.name != Path(__file__).name
0038         and ".git" not in path.parts
0039     )
0040     chunks: list[str] = []
0041     used = 0
0042     for path in files:
0043         relative = path.relative_to(ROOT)
0044         try:
0045             text = path.read_text(encoding="utf-8")
0046         except UnicodeDecodeError:
0047             continue
0048         chunk = f"\n--- {relative} ---\n{text}\n"
0049         if used + len(chunk) > max_chars:
0050             chunks.append(f"\n--- {relative} ---\n[omitted: context limit reached]\n")
0051             break
0052         chunks.append(chunk)
0053         used += len(chunk)
0054     return "".join(chunks) or "[project has no readable files]"
0055

0057 def ask_model(base_url: str, model: str, request: str, context: str) -> str:
0058     payload = {
0059         "model": model,
0060         "temperature": 0.2,
0061         "max_tokens": 1200,
0062         "messages": [
0063             {"role": "system", "content": SYSTEM_PROMPT},
0064             {"role": "user", "content": f"Project context:\n{context}\n\nUser request:\n{request}"},
0065         ],
0066     }
0067     endpoint = base_url.rstrip("/") + "/chat/completions"
0068     http_request = Request(
0069         endpoint,
0070         data=json.dumps(payload).encode("utf-8"),
0071         headers={"Content-Type": "application/json"},
0072         method="POST",
0073     )
0074     with urlopen(http_request, timeout=120) as response:
0075         result = json.load(response)
0076     return result["choices"][0]["message"]["content"]

0079 def main() -> int:
0080     parser = argparse.ArgumentParser(description=__doc__)
0081     parser.add_argument("request", nargs="+", help="Question or requirements task")
0082     parser.add_argument("--url", default=DEFAULT_URL, help="LM Studio OpenAI-compatible base URL")
0083     parser.add_argument("--model", default=DEFAULT_MODEL, help="Model ID served by LM Studio")
0084     parser.add_argument("--max-context", type=int, default=120_000, help="Maximum context characters")
0085     args = parser.parse_args()
0086 
0087     request_text = " ".join(args.request)
0088     try:
0089         answer = ask_model(args.url, args.model, request_text, project_context(args.max_context))
0090     except Exception as error:
0091         parser.error(f"model request failed: {error}")
0092     timestamp = dt.datetime.now(dt.timezone.utc).replace(microsecond=0).isoformat()
0093     CALL_LOG.write_text(
0094         f"# Last local model call\n\n"
0095         f"- **UTC time:** `{timestamp}`\n"
0096         f"- **Model:** `{args.model}`\n"
0097         f"- **Endpoint:** `{args.url.rstrip('/')}`\n"
0098         f"- **Request:** {request_text}\n",
0099         encoding="utf-8",
0100     )
0101     print(f"[local model call: {timestamp} UTC | {args.model}]")
0102     print(answer)
0103     return 0
